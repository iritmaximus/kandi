\begin{subsection}{Positiivinen mitta}
\begin{definition}
	Olkoon $X$ joukko, jolla on $\sigma$-algebra $\Gamma$. Kuvaus $\mu: \Gamma \rightarrow [r, +\infty]$ on joukon $X$ mitta, jos
	\begin{enumerate}
		\item $\mu (\emptyset) = 0$
		\item $A_i \in \Gamma, i \in \N$, erilliset joukot $\implies \mu( \bigcup_{i=1}^{\infty} A_i ) = \Sigma_{i \in \N} \mu(A_i)$
	\end{enumerate}
	Kolmikkoa $(X, \Gamma, \mu)$ kutsutaan mitta-avaruudeksi.
\end{definition}
\end{subsection}
